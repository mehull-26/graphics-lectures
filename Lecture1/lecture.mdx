export const InteractiveCounter = () => {
  const [count, setCount] = React.useState(0);
  
  return (
    <div style={{
      padding: "20px",
      border: "2px solid #4CAF50",
      borderRadius: "8px",
      textAlign: "center",
      margin: "20px 0"
    }}>
      <h3>Renders: {count}</h3>
      <button 
        onClick={() => setCount(count + 1)}
        style={{
          padding: "10px 20px",
          fontSize: "16px",
          backgroundColor: "#4CAF50",
          color: "white",
          border: "none",
          borderRadius: "5px",
          cursor: "pointer"
        }}
      >
        Click to Re-render
      </button>
    </div>
  );
};
